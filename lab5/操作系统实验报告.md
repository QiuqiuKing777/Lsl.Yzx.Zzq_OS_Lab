```c++
int copy_range(pde_t *to, pde_t *from, uintptr_t start, uintptr_t end,
               bool share)
{

    assert(USER_ACCESS(start, end));
    // copy content by page unit.
    do
    {
        // call get_pte to find process A's pte according to the addr start
        pte_t *ptep = get_pte(from, start, 0), *nptep;
        if (ptep == NULL)
        {
            start = ROUNDDOWN(start + PTSIZE, PTSIZE);
            continue;
        }
        // call get_pte to find process B's pte according to the addr start. If
        // pte is NULL, just alloc a PT
        if (*ptep & PTE_V)
        {
            if ((nptep = get_pte(to, start, 1)) == NULL)
            {
                return -E_NO_MEM;
            }
            uint32_t perm = (*ptep & PTE_USER);
            // get page from ptep
            struct Page *page = pte2page(*ptep);
            // alloc a page for process B
            struct Page *npage = alloc_page();
            assert(page != NULL);
            assert(npage != NULL);
            int ret = 0;
            /* LAB5:EXERCISE2 2312796
             */
            // (1) kernel virtual address of src page
            uintptr_t *src_kvaddr = page2kva(page);
            // (2) kernel virtual address of new page
            uintptr_t *dst_kvaddr = page2kva(npage);

            // (3) copy one page content
            memcpy(dst_kvaddr, src_kvaddr, PGSIZE);

            assert(ret == 0);
        }
        start += PGSIZE;
    } while (start != 0 && start < end);
```

#### 问题：如何设计实现`Copy on Write`机制？给出概要设计

​	在创建子进程时不立即复制父进程的内存页，而是让父子进程先共享同一批物理页。为了保证共享期间不发生相互覆盖，内核把这些原本可写的页在父子两边都暂时改成只读，并在页表里做一个“写时复制”的标记，同时维护物理页被多少进程共享的计数。这样父子进程可以正常读取共享页；当任意一方第一次尝试写入某个共享页时，处理器会因为只读保护触发写入异常，内核在异常处理中检查到该页带有写时复制标记后，根据共享计数决定动作：如果该页仍被多个进程共享，就分配一个新的物理页，把原内容复制过去，并让当前进程的映射指向新页且恢复为可写；如果该页已经只被当前进程独占，就无需复制，直接把它恢复为可写即可。进程退出或解除映射时相应减少共享计数，计数归零后再回收物理页。通过这种“先共享、后按需复制”的策略，既保证正确性，又显著降低 fork 的时间和内存开销。

```c++
    if (setup_kstack(proc) != 0) {
        goto bad_fork_cleanup_proc;
    }
    if (copy_mm(clone_flags, proc) != 0) {
        goto bad_fork_cleanup_kstack;
    }
    copy_thread(proc, stack, tf);
    bool intr_flag;
    local_intr_save(intr_flag);  
    proc->pid = get_pid();
    hash_proc(proc);
    set_links(proc);
    local_intr_restore(intr_flag);
    wakeup_proc(proc);
    ret = proc->pid;
fork_out:
    return ret;
bad_fork_cleanup_kstack:
    put_kstack(proc);
bad_fork_cleanup_proc:
    kfree(proc);
    goto fork_out;
}
```

​	`do_fork` 函数是操作系统中创建新进程的核心实现，它负责完整地复制当前进程来生成一个子进程。该函数首先检查系统进程数量限制，然后为子进程分配进程控制块并设置其父进程为当前进程。接着，它通过 `setup_kstack` 为子进程分配内核栈，通过 `copy_mm` 复制或共享内存空间，并通过 `copy_thread` 设置陷阱帧和上下文来确保子进程能够正确开始执行。在完成这些资源分配后，函数会为子进程分配PID并将其加入进程哈希表和管理链表，最后唤醒子进程使其进入就绪状态。


​	`do_execve` 函数是执行新程序的核心实现，它负责将当前进程的内存空间完全替换为新的程序映像。该函数首先进行安全性检查，验证用户传入的程序名称和参数的合法性，并复制程序名称到内核空间。接着，它会清理进程现有的内存管理结构，如果存在原有内存映射则通过 `exit_mmap` 和 `mm_destroy` 等函数释放所有相关资源。然后调用 `load_icode` 函数来实际加载和解析可执行文件二进制代码，建立新的代码段、数据段和堆栈映射。如果程序加载成功，函数会设置新的进程名称并返回，让进程开始执行新程序；如果加载失败，则通过 `do_exit` 终止进程并返回错误码。

​	`sys_wait` 函数是进程等待机制的系统调用接口，它允许父进程等待其子进程的终止并回收系统资源。该函数接收两个参数：要等待的特定子进程的PID（如果为-1则表示等待任意子进程），以及一个用于存储子进程退出状态码的指针。它通过调用 `do_wait` 函数来实际执行等待操作。

###### do_wait函数

```c++
int do_wait(int pid, int *code_store)
{
    struct mm_struct *mm = current->mm;
    if (code_store != NULL)
    {
        if (!user_mem_check(mm, (uintptr_t)code_store, sizeof(int), 1))
        {
            return -E_INVAL;
        }
    }
    struct proc_struct *proc;
    bool intr_flag, haskid;
repeat:
    haskid = 0;
    if (pid != 0)
    {
        proc = find_proc(pid);
        if (proc != NULL && proc->parent == current)
        {
            haskid = 1;
            if (proc->state == PROC_ZOMBIE)
            {
                goto found;
            }
        }
    }
    else
    {
        proc = current->cptr;
        for (; proc != NULL; proc = proc->optr)
        {
            haskid = 1;
            if (proc->state == PROC_ZOMBIE)
            {
                goto found;
            }
        }
    }
    if (haskid)
    {
        current->state = PROC_SLEEPING;
        current->wait_state = WT_CHILD;
        schedule();
        if (current->flags & PF_EXITING)
        {
            do_exit(-E_KILLED);
        }
        goto repeat;
    }
    return -E_BAD_PROC;
found:
    if (proc == idleproc || proc == initproc)
    {
        panic("wait idleproc or initproc.\n");
    }
    if (code_store != NULL)
    {
        *code_store = proc->exit_code;
    }
    local_intr_save(intr_flag);
    {
        unhash_proc(proc);
        remove_links(proc);
    }
    local_intr_restore(intr_flag);
    put_kstack(proc);
    kfree(proc);
    return 0;
}
```

​	`do_wait` 函数负责父进程等待和回收子进程资源。它首先检查子进程状态：如果指定PID或任意子进程已处于僵尸状态，则立即回收其内核栈和进程控制块，并返回退出码；如果子进程仍在运行，则让父进程进入睡眠等待状态，稍后被唤醒时重新检查。这确保了父进程能正确同步子进程结束，防止僵尸进程产生。

###### do_exit函数

```c++
int do_exit(int error_code)
{
    if (current == idleproc)
    {
        panic("idleproc exit.\n");
    }
    if (current == initproc)
    {
        panic("initproc exit.\n");
    }
    struct mm_struct *mm = current->mm;
    if (mm != NULL)
    {
        lsatp(boot_pgdir_pa);
        if (mm_count_dec(mm) == 0)
        {
            exit_mmap(mm);
            put_pgdir(mm);
            mm_destroy(mm);
        }
        current->mm = NULL;
    }
    current->state = PROC_ZOMBIE;
    current->exit_code = error_code;
    bool intr_flag;
    struct proc_struct *proc;
    local_intr_save(intr_flag);
    {
        proc = current->parent;
        if (proc->wait_state == WT_CHILD)
        {
            wakeup_proc(proc);
        }
        while (current->cptr != NULL)
        {
            proc = current->cptr;
            current->cptr = proc->optr;
            proc->yptr = NULL;
            if ((proc->optr = initproc->cptr) != NULL)
            {
                initproc->cptr->yptr = proc;
            }
            proc->parent = initproc;
            initproc->cptr = proc;
            if (proc->state == PROC_ZOMBIE)
            {
                if (initproc->wait_state == WT_CHILD)
                {
                    wakeup_proc(initproc);
                }
            }
        }
    }
    local_intr_restore(intr_flag);
    schedule();
    panic("do_exit will not return!! %d.\n", current->pid);
}

```

​	`do_exit` 函数处理进程退出。它释放进程内存资源，将状态设为僵尸进程，保存退出码并唤醒等待的父进程。同时将子进程过继给初始化进程，最后调度其他进程运行，完成进程生命周期管理。

##### 问题回答

（1）在执行流程中，子进程调用`fork、exec、wait、exit`以及到`ecall`都是在用户态进行的。而从内核trap入口到分发系统调用（trap_dispatch）再到上面分析的这些sys、do函数，最后到将返回值写入中断帧（trapframe）都是在内核状态进行的。最后在调用sret后，又重新返回用户态。

（2）内核态到用户态是通过`sret`实现的，而用户态到内核态是通过`ecall`实现的。

### （五）扩展Challenge

#### 1.实现 Copy on Write （COW）机制

##### 设计思路

1. fork 时不再复制物理页，父子进程共享同一物理页；

2. 对原本可写的用户页，父子双方都清掉 PTE_W 并打上 PTE_COW 标记；

3. 当子/父第一次写该页时触发Store page fault，在异常里检测 PTE_COW：

- 若 ref>1：分配新页、拷贝、改成可写、清掉 COW
- 若 ref==1：直接改回可写、清掉 COW

##### 代码实现

###### `riscv.h`：在该文件中增加软件位

###### `vmm.c`：在复制时，实现“共享+COW”路径

###### `pmm.c`：在 copy_range 里实现 COW 共享映射

​	我们在原来的EXERCISE2的位置实现，原来传进来的参数中就存在这share，在这里我们把share应用起来，改成下面这样的代码：

```c++

* nage with the linear addr start
*
* Some Useful MACROs and DEFINEs, you can use them in below

* MACROs or Functions:
*    page2kva(struct Page *page): return the kernel vritual addr of

// Only writable pages need COW. Read-only pages can be shared directly.
if (perm & PTE_W) {
    perm = (perm & ~PTE_W) | PTE_COW;
    // update parent's mapping to be COW as well
    *ptep = (*ptep & ~PTE_W) | PTE_COW;
    tlb_invalidate(from, start);

ret = page_insert(to, page, start, perm);
} else {
```

###### `trap.c`：在 Store Page Fault 里做“写时复制”

​	在这里我们找到`case CAUSE_STORE_PAGE_FAULT:`分支，然后将其代码改成这个形式即可：

```c++
case CAUSE_STORE_PAGE_FAULT:{
    uintptr_t badaddr = read_csr(stval);
    uintptr_t la = ROUNDDOWN(badaddr, PGSIZE);

    if (current != NULL && current->mm != NULL) {
        pte_t *ptep = get_pte(current->mm->pgdir, badaddr, 0);
        if (ptep != NULL && (*ptep & PTE_V) && (*ptep & PTE_COW)) {
            struct Page *page = pte2page(*ptep);

            if (page_ref(page) > 1) {
                struct Page *npage = alloc_page();
                if (npage == NULL) panic("COW: out of memory");
                memcpy(page2kva(npage), page2kva(page), PGSIZE);

                uint32_t perm = (*ptep & PTE_USER);
                perm = (perm | PTE_W) & ~PTE_COW;
                page_insert(current->mm->pgdir, npage, la, perm);
            } else {
                *ptep = (*ptep | PTE_W) & ~PTE_COW;
                tlb_invalidate(current->mm->pgdir, la);
            }
            break; // handled
        }
    }
    cprintf("Store/AMO page fault\n");
    break;
}
```

​	这里是我们实现COW的核心部分，我们的实现逻辑是：

​	首先使用`read_csr(stval)` 取 fault 地址（RISC-V 的 badaddr），然后判断 `PTE_COW` 才是 COW fault；否则仍按原逻辑报错。接着根据ref的大小来进行判断，若`ref>1`：复制新页后 `page_insert` 会替换映射，并把旧页 ref--；若`ref==1`：不用复制，直接把页改回可写即可，最后 `tlb_invalidate` 保证新权限生效。

#### 说明该用户程序是何时被预先加载到内存中的？与我们常用操作系统的加载有何区别，原因是什么？

##### “预先”加载

​	用户程序在 `exec` 阶段（`load_icode`）被提前装入内存，而不是在 `RUNNING `后再加载，这样在程序执行之前所有需要的内容就已经被加载到内存中了。

##### 与常用操作系统的加载的异同

常用 `OS`（`Linux/Windows` 等）里：

+ `execve` 时从**磁盘文件系统**读取可执行文件；

+ 往往是**按需加载/缺页再读**（`demand paging`），并不一定一次性把所有段拷进物理内存；

+ 文件页还能用` page cache `复用。

uCore ：

+ 用户程序通常是**编译后直接打包进内核镜像/`ramdisk`**，内核从内存里拿到 ELF，不经过真实磁盘文件系统。
+ `load_icode()` 直接为 ELF 的各个段分配物理页并 `memcpy` 进去， **基本是一次性装完**，没有复杂的按需分页/缓存。

##### 原因

+ `ucore`没有完整的磁盘驱动 + 文件系统支持
+ **虚拟内存与按需加载机制尚未完整实现**，在实验阶段更容易用“全量拷贝段到内存”来保证可控、可调试。
